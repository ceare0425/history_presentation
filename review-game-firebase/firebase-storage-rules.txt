// Firebase Storage 보안 규칙 - 국립중앙박물관 스탬프 투어(유물 사진 업로드)용
// Firebase 콘솔 → Storage → 규칙(Rules) 탭에 이 내용을 붙여넣고 "게시".
// (Realtime Database 규칙과는 별개입니다. 그건 firebase-database-rules.json 을 DB 규칙 탭에.)
//
// 경로: museum/{투어코드}/{학년-반-번호}/{유물id}.jpg
// - 읽기: 공개 (교사용 현황판이 학생 사진 썸네일을 보기 위함)
// - 쓰기: 3MB 이하 이미지 파일만 허용 (그 외 인증 없음 — 교실용, 복습 게임 규칙과 동일한 개방 정책)

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /museum/{tour}/{sid}/{fileName} {
      allow read: if true;
      allow write: if request.resource != null
                   && request.resource.size < 3 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*');
      allow delete: if false;
    }
  }
}
